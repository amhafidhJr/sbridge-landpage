SMS Bridge Gateway

Integration guide for connecting your backend system with the SMS Bridge Android App.

Laravel Integration Guide

Overview

SMS Bridge Gateway allows your Laravel backend to send SMS messages through an Android device running the SMS Bridge App. This guide provides a complete setup: database models, controllers, routes, and integration steps.

Requirements

  • Laravel 8+
  • PHP 8.0+
  • MySQL/PostgreSQL database
  • Android device with SMS Bridge App installed

Step 1: Database Models

SMS Model

Create a model Sms:

<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Sms extends Model
{
    use HasFactory;

    protected $fillable = [
        'sms_status',
        'message',
        'error_message',
        'sent_at',
        'delivered_at',
        'user_phone_number_id'
    ];

    const STATUS_QUEUED    = 'QUEUED';
    const STATUS_PENDING   = 'PENDING';
    const STATUS_SENT      = 'SENT';
    const STATUS_DELIVERED = 'DELIVERED';
    const STATUS_FAILED    = 'FAILED';

    public static function getValidStatuses(): array
    {
        return [
            self::STATUS_QUEUED,
            self::STATUS_PENDING,
            self::STATUS_SENT,
            self::STATUS_DELIVERED,
            self::STATUS_FAILED,
        ];
    }

    public function recipientNumber()
    {
        return $this->belongsTo(UserPhoneNumber::class, 'user_phone_number_id');
    }
}

Step 2: Controller

<?php
namespace App\Http\Controllers;

use App\Models\Sms;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Validator;

class SmsBridgeController extends Controller
{
    public function getPendingSms()
    {
        $pendingSms = Sms::with('recipientNumber')
            ->whereIn('sms_status', ['QUEUED', 'FAILED'])
            ->orderBy('id', 'asc')
            ->take(50)
            ->get();

        $response = $pendingSms->map(function ($sms) {
            return [
                'id'              => $sms->id,
                'userId'          => $sms->user_phone_number_id,
                'recipientNumber' => $sms->recipientNumber->phone_number ?? null,
                'message'         => $sms->message,
                'status'          => $sms->sms_status,
                'sentAt'          => $sms->sent_at,
                'deliveredAt'     => $sms->delivered_at,
            ];
        });

        return response()->json($response);
    }

    public function testConnection()
    {
        return response()->json(['status' => 'ok', 'message' => 'Connection successful']);
    }

    public function updateStatus(Request $request)
    {
        $validator = Validator::make($request->all(), [
            'message_id' => 'required|string',
            'status'     => 'required|string|in:' . implode(',', Sms::getValidStatuses()),
        ]);

        if ($validator->fails()) {
            return response()->json(['success' => false, 'errors' => $validator->errors()], 422);
        }

        $sms = Sms::find($request->message_id);
        if (!$sms) return response()->json(['success' => false, 'message' => 'SMS not found'], 404);

        $sms->sms_status = $request->status;
        if ($request->status === Sms::STATUS_SENT) $sms->sent_at = now();
        if ($request->status === Sms::STATUS_DELIVERED) $sms->delivered_at = now();
        if ($request->status === Sms::STATUS_FAILED && $request->has('error_message'))
            $sms->error_message = $request->error_message;
        $sms->save();

        return response()->json(['success' => true, 'message' => 'Status updated successfully']);
    }
}

    Route::get('/pending-sms', [SmsBridgeController::class, 'getPendingSms']);

Integration Notes

  • The Android app polls /pending-sms to fetch messages with status QUEUED or FAILED
  • After sending, the app posts status updates to /update-status
  • Limit messages fetched per request (e.g., 50) to prevent overloading the device
  • Handle FAILED messages carefully and log errors

db = SQLAlchemy()

class UserPhoneNumber(db.Model):
    __tablename__ = 'user_phone_numbers'

    id           = db.Column(db.Integer, primary_key=True)
    phone_number = db.Column(db.String(20), nullable=False)

    def __repr__(self):
        return f'<UserPhoneNumber {self.phone_number}>'


class Sms(db.Model):
    __tablename__ = 'sms'

    STATUS_QUEUED    = 'QUEUED'
    STATUS_PENDING   = 'PENDING'
    STATUS_SENT      = 'SENT'
    STATUS_DELIVERED = 'DELIVERED'
    STATUS_FAILED    = 'FAILED'

    VALID_STATUSES = [STATUS_QUEUED, STATUS_PENDING, STATUS_SENT, STATUS_DELIVERED, STATUS_FAILED]

    id                  = db.Column(db.Integer, primary_key=True)
    sms_status          = db.Column(db.String(20), nullable=False, default='QUEUED')
    message             = db.Column(db.Text, nullable=False)
    error_message       = db.Column(db.Text, nullable=True)
    sent_at             = db.Column(db.DateTime, nullable=True)
    delivered_at        = db.Column(db.DateTime, nullable=True)
    user_phone_number_id = db.Column(db.Integer, db.ForeignKey('user_phone_numbers.id'), nullable=False)
    created_at          = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at          = db.Column(db.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    def to_dict(self):
        return {
            'id':              self.id,
            'userId':          self.user_phone_number_id,
            'recipientNumber': self.recipient_number.phone_number if self.recipient_number else None,
            'message':         self.message,
            'status':          self.sms_status,
            'sentAt':          self.sent_at.isoformat() if self.sent_at else None,
            'deliveredAt':     self.delivered_at.isoformat() if self.delivered_at else None,
        }

Step 3: Flask App & Routes

# app.py
from flask import Flask, jsonify, request
from models import db, Sms, UserPhoneNumber
from datetime import datetime
import logging

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://user:password@localhost/sms_bridge'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db.init_app(app)
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


@app.route('/test-connection', methods=['GET'])
def test_connection():
    logger.info('Test connection endpoint hit')
    return jsonify({'status': 'ok', 'message': 'Connection successful'})


@app.route('/api/v1/sms-bridge/pending-sms', methods=['GET'])
def get_pending_sms():
    logger.info('Fetching pending SMS (QUEUED and FAILED)')

    pending = Sms.query \
        .filter(Sms.sms_status.in_(['QUEUED', 'FAILED'])) \
        .order_by(Sms.id.asc()) \
        .limit(50) \
        .all()

    logger.info(f'Found {len(pending)} pending SMS records')
    return jsonify([sms.to_dict() for sms in pending])


@app.route('/api/v1/sms-bridge/update-status', methods=['POST'])
def update_status():
    data = request.get_json()

    if not data:
        return jsonify({'success': False, 'message': 'Invalid JSON body'}), 400

    message_id = data.get('message_id')
    status     = data.get('status')

    # Validate required fields
    if not message_id or not status:
        return jsonify({'success': False, 'message': 'message_id and status are required'}), 422

    if status not in Sms.VALID_STATUSES:
        return jsonify({
            'success': False,
            'message': f'Invalid status. Must be one of: {", ".join(Sms.VALID_STATUSES)}'
        }), 422

    sms = Sms.query.get(message_id)
    if not sms:
        return jsonify({'success': False, 'message': 'SMS record not found'}), 404

    logger.info(f'Updating SMS {message_id} from {sms.sms_status} to {status}')

    sms.sms_status = status

    if status == Sms.STATUS_SENT:
        sms.sent_at = datetime.utcnow()
    elif status == Sms.STATUS_DELIVERED:
        sms.delivered_at = datetime.utcnow()
    elif status == Sms.STATUS_FAILED:
        sms.error_message = data.get('error_message')

    db.session.commit()

    return jsonify({
        'success': True,
        'message': 'Status updated successfully',
        'data': sms.to_dict()
    })


if __name__ == '__main__':
    with app.app_context():
        db.create_all()
    app.run(debug=True, port=5000)

package com.example.smsbridge.entity;

import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "sms")
@Data
@NoArgsConstructor
@AllArgsConstructor
public class Sms {

    public enum Status { QUEUED, PENDING, SENT, DELIVERED, FAILED }

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Enumerated(EnumType.STRING)
    @Column(name = "sms_status", nullable = false)
    private Status smsStatus = Status.QUEUED;

    @Column(nullable = false, columnDefinition = "TEXT")
    private String message;

    @Column(name = "error_message", columnDefinition = "TEXT")
    private String errorMessage;

    @Column(name = "sent_at")
    private LocalDateTime sentAt;

    @Column(name = "delivered_at")
    private LocalDateTime deliveredAt;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "user_phone_number_id", nullable = false)
    private UserPhoneNumber userPhoneNumber;
}

Step 6: Controller

// SmsBridgeController.java
package com.example.smsbridge.controller;

import com.example.smsbridge.dto.*;
import com.example.smsbridge.entity.Sms;
import com.example.smsbridge.repository.SmsRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.PageRequest;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.time.LocalDateTime;
import java.util.*;
import java.util.stream.Collectors;

@Slf4j
@RestController
@RequiredArgsConstructor
public class SmsBridgeController {

    private final SmsRepository smsRepository;

    @GetMapping("/test-connection")
    public ResponseEntity<Map<String, String>> testConnection() {
        log.info("=== TEST CONNECTION ENDPOINT HIT ===");
        return ResponseEntity.ok(Map.of("status", "ok", "message", "Connection successful"));
    }

    @GetMapping("/api/v1/sms-bridge/pending-sms")
    public ResponseEntity<List<SmsPendingResponse>> getPendingSms() {
        log.info("Fetching pending SMS (QUEUED and FAILED)");

        List<Sms> pending = smsRepository.findBySmsStatusInOrderByIdAsc(
            List.of(Sms.Status.QUEUED, Sms.Status.FAILED),
            PageRequest.of(0, 50)
        );

        List<SmsPendingResponse> response = pending.stream().map(sms -> {
            SmsPendingResponse dto = new SmsPendingResponse();
            dto.setId(sms.getId());
            dto.setUserId(sms.getUserPhoneNumber().getId());
            dto.setRecipientNumber(sms.getUserPhoneNumber().getPhoneNumber());
            dto.setMessage(sms.getMessage());
            dto.setStatus(sms.getSmsStatus().name());
            dto.setSentAt(sms.getSentAt());
            dto.setDeliveredAt(sms.getDeliveredAt());
            return dto;
        }).collect(Collectors.toList());

        log.info("Found {} pending SMS records", response.size());
        return ResponseEntity.ok(response);
    }

    @PostMapping("/api/v1/sms-bridge/update-status")
    public ResponseEntity<Map<String, Object>> updateStatus(@RequestBody UpdateStatusRequest req) {
        if (req.getMessage_id() == null || req.getStatus() == null) {
            return ResponseEntity.badRequest().body(Map.of(
                "success", false, "message", "message_id and status are required"
            ));
        }

        Sms.Status newStatus;
        try {
            newStatus = Sms.Status.valueOf(req.getStatus());
        } catch (IllegalArgumentException e) {
            return ResponseEntity.unprocessableEntity().body(Map.of(
                "success", false, "message", "Invalid status value"
            ));
        }

        Optional<Sms> optSms = smsRepository.findById(Long.parseLong(req.getMessage_id()));
        if (optSms.isEmpty()) {
            return ResponseEntity.status(404).body(Map.of("success", false, "message", "SMS record not found"));
        }

        Sms sms = optSms.get();
        log.info("Updating SMS {} from {} to {}", sms.getId(), sms.getSmsStatus(), newStatus);

        sms.setSmsStatus(newStatus);
        switch (newStatus) {
            case SENT      -> sms.setSentAt(LocalDateTime.now());
            case DELIVERED -> sms.setDeliveredAt(LocalDateTime.now());
            case FAILED    -> sms.setErrorMessage(req.getError_message());
        }
        smsRepository.save(sms);

        return ResponseEntity.ok(Map.of("success", true, "message", "Status updated successfully"));
    }
}

Integration Notes

  • Use @CrossOrigin annotation or a global CorsFilter bean if calling from the Android app across origins
  • The @Slf4j Lombok annotation provides the log variable automatically
  • For Gradle, replace the Maven dependencies with the equivalent implementation entries in build.gradle

        tableName: 'sms',
        timestamps: true,

Step 5: Controller

// controllers/SmsBridgeController.js
const { Sms, UserPhoneNumber } = require('../models');

exports.testConnection = (req, res) => {
    console.log('=== TEST CONNECTION ENDPOINT HIT ===');
    res.json({ status: 'ok', message: 'Connection successful' });
};

exports.getPendingSms = async (req, res) => {
    try {
        console.log('Fetching pending SMS (QUEUED and FAILED)');

        const pending = await Sms.findAll({
            where: { sms_status: ['QUEUED', 'FAILED'] },
            include: [{ model: UserPhoneNumber, as: 'recipientNumber' }],
            order: [['id', 'ASC']],
            limit: 50,
        });

        console.log(`Found ${pending.length} pending SMS records`);

        const response = pending.map(sms => ({
            id:              sms.id,
            userId:          sms.user_phone_number_id,
            recipientNumber: sms.recipientNumber?.phone_number ?? null,
            message:         sms.message,
            status:          sms.sms_status,
            sentAt:          sms.sent_at,
            deliveredAt:     sms.delivered_at,
        }));

        res.json(response);
    } catch (error) {
        console.error('Error fetching pending SMS:', error);
        res.status(500).json({ success: false, message: 'Internal server error' });
    }
};

exports.updateStatus = async (req, res) => {
    const { message_id, status, error_message } = req.body;

    if (!message_id || !status) {
        return res.status(422).json({
            success: false,
            message: 'message_id and status are required',
        });
    }

    if (!Sms.VALID_STATUSES.includes(status)) {
        return res.status(422).json({
            success: false,
            message: `Invalid status. Must be one of: ${Sms.VALID_STATUSES.join(', ')}`,
        });
    }

    try {
        const sms = await Sms.findByPk(message_id);
        if (!sms) {
            return res.status(404).json({ success: false, message: 'SMS record not found' });
        }

        console.log(`Updating SMS ${message_id} from ${sms.sms_status} to ${status}`);

        const updates = { sms_status: status };
        if (status === 'SENT')      updates.sent_at      = new Date();
        if (status === 'DELIVERED') updates.delivered_at = new Date();
        if (status === 'FAILED' && error_message) updates.error_message = error_message;

        await sms.update(updates);

        res.json({ success: true, message: 'Status updated successfully' });
    } catch (error) {
        console.error('Error updating SMS status:', error);
        res.status(500).json({ success: false, message: 'Internal server error' });
    }
};

app.get('/test-connection', ctrl.testConnection);
app.use('/api/v1/sms-bridge', smsBridgeRoutes);

const PORT = process.env.PORT || 3000;

sequelize.sync().then(() => {
    console.log('Database connected and synced');
    app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
}).catch(err => console.error('DB connection failed:', err));

Integration Notes

  • Use cors npm package to allow requests from the Android app if needed
  • For production, use pm2 or a Docker container to keep the server alive
  • Consider adding request validation middleware like express-validator
