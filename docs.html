SMS Bridge Gateway

Integration guide for connecting your backend system with the SMS Bridge Android App.

Laravel Integration Guide

Overview

SMS Bridge Gateway allows your Laravel backend to send SMS messages through an Android device running the SMS Bridge App. This guide provides a complete setup: database models, controllers, routes, and integration steps.

Requirements

  • Laravel 8+
  • PHP 8.0+
  • MySQL/PostgreSQL database
  • Android device with SMS Bridge App installed

Step 1: Database Models

SMS Model

Create a model Sms:

<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Sms extends Model
{
    use HasFactory;

    protected $fillable = [
        'sms_status',
        'message',
        'error_message',
        'sent_at',
        'delivered_at',
        'user_phone_number_id'
    ];

    // Status values - aligned with Android app
    const STATUS_QUEUED = 'QUEUED';
    const STATUS_PENDING = 'PENDING';
    const STATUS_SENT = 'SENT';
    const STATUS_DELIVERED = 'DELIVERED';
    const STATUS_FAILED = 'FAILED';

    public static function getValidStatuses(): array
    {
        return [
            self::STATUS_QUEUED,
            self::STATUS_PENDING,
            self::STATUS_SENT,
            self::STATUS_DELIVERED,
            self::STATUS_FAILED,
        ];
    }

    public function recipientNumber()
    {
        return $this->belongsTo(UserPhoneNumber::class, 'user_phone_number_id');
    }
}
UserPhoneNumber Model

Create a model UserPhoneNumber:

<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;

class UserPhoneNumber extends Model

Step 2: Controller

Create a controller SmsBridgeController:

<?php

namespace App\Http\Controllers;

use App\Models\Sms;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Validator;

class SmsBridgeController extends Controller
{
    // GET /pending-sms - Get all QUEUED and FAILED SMS
    public function getPendingSms()
    {
        Log::info('Fetching pending SMS (QUEUED and FAILED)');

        $pendingSms = Sms::with('recipientNumber')
            ->whereIn('sms_status', ['QUEUED', 'FAILED'])
            ->orderBy('id', 'asc')
            ->take(50)
            ->get();

        Log::info('Found ' . $pendingSms->count() . ' pending SMS records');

        $response = $pendingSms->map(function ($sms) {
            return [
                'id'              => $sms->id,
                'userId'          => $sms->user_phone_number_id,
                'recipientNumber' => $sms->recipientNumber->phone_number ?? null,
                'message'         => $sms->message,
                'status'          => $sms->sms_status,
                'sentAt'          => $sms->sent_at,
                'deliveredAt'     => $sms->delivered_at,
            ];
        });

        return response()->json($response);
    }

    public function testConnection()
    {
        Log::info('=== TEST CONNECTION ENDPOINT HIT ===');

        return response()->json([
            'status'  => 'ok',
            'message' => 'Connection successful',
        ]);
    }

    public function updateStatus(Request $request)
    {
        Log::info('SMS status update request received', $request->all());

        $validator = Validator::make($request->all(), [
            'message_id' => 'required|string',
            'status'     => 'required|string|in:' . implode(',', Sms::getValidStatuses()),
        ]);

        if ($validator->fails()) {
            Log::warning('SMS status update validation failed', ['errors' => $validator->errors()->toArray()]);
            return response()->json(['success' => false, 'message' => 'Invalid request data', 'errors' => $validator->errors()], 422);
        }

        try {
            $messageId = $request->message_id;
            $status    = $request->status;

            $sms = Sms::where('id', $messageId)->first();

            if (!$sms) {
                Log::warning('SMS record not found', ['message_id' => $messageId]);
                return response()->json(['success' => false, 'message' => 'SMS record not found'], 404);
            }

            Log::info('Updating SMS status', ['message_id' => $messageId, 'from_status' => $sms->sms_status, 'to_status' => $status]);

            $sms->sms_status = $status;

            switch ($status) {
                case Sms::STATUS_SENT:
                    $sms->sent_at = now();
                    break;
                case Sms::STATUS_DELIVERED:
                    $sms->delivered_at = now();
                    break;
                case Sms::STATUS_FAILED:
                    if ($request->has('error_message')) {
                        $sms->error_message = $request->error_message;
                    }
                    break;
            }

            $sms->save();

            Log::info('SMS status updated successfully', ['message_id' => $messageId, 'status' => $status]);

            return response()->json([
                'success' => true,
                'message' => 'Status updated successfully',
                'data'    => [
                    'message_id'   => $sms->id,
                    'status'       => $sms->sms_status,
                    'sent_at'      => $sms->sent_at,
                    'delivered_at' => $sms->delivered_at,
                ],
            ]);

        } catch (\Exception $e) {
            Log::error('Error updating SMS status', ['error' => $e->getMessage()]);
            return response()->json(['success' => false, 'message' => 'Internal server error'], 500);
        }
    }
}

Step 3: Routes

Add these routes in routes/api.php:

use App\Http\Controllers\SmsBridgeController;

Route::prefix('api/v1/sms-bridge')->group(function () {
    Route::get('/pending-sms', [SmsBridgeController::class, 'getPendingSms']);
    Route::post('/update-status', [SmsBridgeController::class, 'updateStatus']);
});

Route::get('/test-connection', [SmsBridgeController::class, 'testConnection']);

Integration Notes

  • The Android app calls /pending-sms to fetch QUEUED or FAILED messages
  • After sending, the app calls /update-status
  • Limit is set to 50 messages per request to avoid overloading the Android device
  • Use a constant BASE_URL in the Android app

Python Integration Guide (FastAPI)

Overview

Complete integration for Python using FastAPI and SQLAlchemy.

Requirements

  • Python 3.9+
  • FastAPI + Uvicorn

Step 1: Models (models.py)

from sqlalchemy import Column, Integer, String, DateTime, ForeignKey
from sqlalchemy.orm import relationship
from database import Base
from datetime import datetime

class Sms(Base):
    __tablename__ = "sms"

    id = Column(Integer, primary_key=True, index=True)
    sms_status = Column(String, default="QUEUED")
    message = Column(String, nullable=False)
    error_message = Column(String, nullable=True)
    sent_at = Column(DateTime, nullable=True)
    delivered_at = Column(DateTime, nullable=True)
    user_phone_number_id = Column(Integer, ForeignKey("user_phone_numbers.id"))

    recipient_number = relationship("UserPhoneNumber", back_populates="sms")

    STATUS_QUEUED = "QUEUED"
    STATUS_PENDING = "PENDING"
    STATUS_SENT = "SENT"
    STATUS_DELIVERED = "DELIVERED"
    STATUS_FAILED = "FAILED"

    @staticmethod
    def get_valid_statuses():
        return [Sms.STATUS_QUEUED, Sms.STATUS_PENDING, Sms.STATUS_SENT, Sms.STATUS_DELIVERED, Sms.STATUS_FAILED]


class UserPhoneNumber(Base):
    __tablename__ = "user_phone_numbers"
    id = Column(Integer, primary_key=True, index=True)
    phone_number = Column(String, unique=True, nullable=False)

    sms = relationship("Sms", back_populates="recipient_number")

Step 2: Router/Controller (sms_bridge.py)

from fastapi import APIRouter, Depends, HTTPException, Request
from sqlalchemy.orm import Session
from sqlalchemy import or_
from datetime import datetime
import logging

from models import Sms, UserPhoneNumber
from database import get_db

router = APIRouter(prefix="/api/v1/sms-bridge")
logger = logging.getLogger(__name__)

@router.get("/pending-sms")
async def get_pending_sms(db: Session = Depends(get_db)):
    logger.info("Fetching pending SMS (QUEUED and FAILED)")
    
    pending_sms = db.query(Sms).join(UserPhoneNumber).filter(
        or_(Sms.sms_status == Sms.STATUS_QUEUED, Sms.sms_status == Sms.STATUS_FAILED)
    ).order_by(Sms.id.asc()).limit(50).all()

    response = []
    for sms in pending_sms:
        response.append({
            'id': sms.id,
            'userId': sms.user_phone_number_id,
            'recipientNumber': sms.recipient_number.phone_number if sms.recipient_number else None,
            'message': sms.message,
            'status': sms.sms_status,
            'sentAt': sms.sent_at.isoformat() if sms.sent_at else None,
            'deliveredAt': sms.delivered_at.isoformat() if sms.delivered_at else None,
        })
    return response

@router.get("/test-connection")
async def test_connection():
    logger.info("=== TEST CONNECTION ENDPOINT HIT ===")
    return {"status": "ok", "message": "Connection successful"}

@router.post("/update-status")
async def update_status(request: Request, db: Session = Depends(get_db)):
    data = await request.json()
    message_id = data.get("message_id")
    status = data.get("status")

    if not message_id or status not in Sms.get_valid_statuses():
        raise HTTPException(status_code=422, detail="Invalid request data")

    sms = db.query(Sms).filter(Sms.id == int(message_id)).first()
    if not sms:
        raise HTTPException(status_code=404, detail="SMS record not found")

    sms.sms_status = status
    if status == Sms.STATUS_SENT:
        sms.sent_at = datetime.utcnow()
    elif status == Sms.STATUS_DELIVERED:
        sms.delivered_at = datetime.utcnow()
    elif status == Sms.STATUS_FAILED:
        sms.error_message = data.get("error_message")

    db.commit()
    db.refresh(sms)

    return {
        "success": True,
        "message": "Status updated successfully",
        "data": {
            "message_id": sms.id,
            "status": sms.sms_status,
            "sent_at": sms.sent_at.isoformat() if sms.sent_at else None,
            "delivered_at": sms.delivered_at.isoformat() if sms.delivered_at else None,
        }
    }

Node.js Integration Guide (Express.js)

Overview

Complete integration using Express.js and Sequelize (or Mongoose).

Step 1: Controller (smsBridgeController.js)

exports.getPendingSms = async (req, res) => {
    const pending = await Sms.findAll({
        where: { sms_status: ['QUEUED', 'FAILED'] },
        include: ['recipientNumber'],
        limit: 50
    });

    res.json(pending.map(sms => ({
        id: sms.id,
        userId: sms.user_phone_number_id,
        recipientNumber: sms.recipientNumber?.phone_number,
        message: sms.message,
        status: sms.sms_status,
        sentAt: sms.sent_at,
        deliveredAt: sms.delivered_at
    })));
};

exports.updateStatus = async (req, res) => {
    const { message_id, status, error_message } = req.body;
    const sms = await Sms.findByPk(message_id);
    if (!sms) return res.status(404).json({success: false, message: "Not found"});

    sms.sms_status = status;
    if (status === 'SENT') sms.sent_at = new Date();
    if (status === 'DELIVERED') sms.delivered_at = new Date();
    if (status === 'FAILED') sms.error_message = error_message;

    await sms.save();
    res.json({success: true, message: "Status updated"});
};
